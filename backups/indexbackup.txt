      /*

      let timerInterval = null;

      let elapsedInterval = null;
      let elapsedStartTime = null;
      let elapsedTimeMs = 0;

      function selectedDurationMs() {
        return (
          Number(document.getElementById("timerDuration").value) * 60 * 1000
        );
      }

      function stopQuizTimer() {
        if (timerInterval !== null) {
          clearInterval(timerInterval);
          timerInterval = null;
        }

        timerDeadline = null;
      }

      function renderTimer() {
        const display = document.getElementById("timerDisplay");
        const stat = document.getElementById("timerStat");
        const mode = document.getElementById("timerMode").value;

        if (mode === "off") {
          display.textContent = "Off";
          stat.classList.remove("urgent");
          document.body.classList.remove("timer-warning");
          return;
        }

        if (mode === "countup") {
          const totalSeconds = Math.floor(elapsedTimeMs / 1000);
          const minutes = Math.floor(totalSeconds / 60);
          const seconds = totalSeconds % 60;

          display.textContent = `${minutes}:${String(seconds).padStart(2, "0")}`;

          stat.classList.remove("urgent");
          document.body.classList.remove("timer-warning");
          return;
        }

        const ms = Math.max(0, remainingTimeMs ?? selectedDurationMs());
        const totalSeconds = Math.ceil(ms / 1000);
        const minutes = Math.floor(totalSeconds / 60);
        const seconds = totalSeconds % 60;

        display.textContent = `${minutes}:${String(seconds).padStart(2, "0")}`;

        const urgent =
          quizStarted && ms > 0 && ms <= 10000 && !quizExpired && !quizFinished;

        stat.classList.toggle("urgent", urgent);
        document.body.classList.toggle("timer-warning", urgent);
      }

      function renderElapsedTimer() {
        const totalSeconds = Math.floor(elapsedTimeMs / 1000);
        const seconds = totalSeconds % 60;
        const totalMinutes = Math.floor(totalSeconds / 60);
        const minutes = totalMinutes % 60;
        const hours = Math.floor(totalMinutes / 60);

        const formatted =
          hours > 0
            ? `${hours}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`
            : `${totalMinutes}:${String(seconds).padStart(2, "0")}`;

        document.getElementById("elapsedDisplay").textContent = formatted;
      }

      function updateCountUpButton() {
        const button = document.getElementById("stopElapsedButton");

        if (!button) return;

        const isCountUp =
          document.getElementById("timerMode").value === "countup";

        const quizActive = quizStarted && !quizFinished && !quizExpired;

        // Only show the button during an active Count-Up quiz.
        button.style.display = isCountUp && quizActive ? "" : "none";

        // Never allow restarting after the quiz has ended.
        button.disabled = !isCountUp || !quizActive;

        // Match the label to the timer's actual state.
        button.textContent =
          elapsedInterval !== null ? "Stop Timer" : "Start Timer";
      }

      function startElapsedTimer() {
        // Do not start or resume after the quiz has ended.
        if (!quizStarted || quizFinished || quizExpired) {
          updateCountUpButton();
          return;
        }

        // Do not create duplicate intervals.
        if (elapsedInterval !== null) {
          updateCountUpButton();
          return;
        }

        // Resume from the saved elapsed time.
        elapsedStartTime = Date.now() - elapsedTimeMs;

        elapsedInterval = setInterval(() => {
          elapsedTimeMs = Date.now() - elapsedStartTime;
          renderElapsedTimer();
          renderTimer();
        }, 200);

        updateCountUpButton();
      }

      function stopElapsedTimer() {
        if (elapsedInterval !== null) {
          clearInterval(elapsedInterval);
          elapsedInterval = null;
        }

        // Save the elapsed time at the moment the timer stops.
        if (elapsedStartTime !== null) {
          elapsedTimeMs = Date.now() - elapsedStartTime;
          elapsedStartTime = null;
        }

        renderElapsedTimer();
        renderTimer();
        updateCountUpButton();
      }

      function stopCountUpTimer() {
        if (!quizStarted || quizFinished || quizExpired) {
          updateCountUpButton();
          return;
        }

        if (document.getElementById("timerMode").value !== "countup") {
          updateCountUpButton();
          return;
        }

        if (elapsedInterval !== null) {
          // Running → stopped.
          stopElapsedTimer();
        } else {
          // Stopped → running, preserving elapsedTimeMs.
          startElapsedTimer();
        }
      }

      function startQuizTimer(
        durationMs = remainingTimeMs ?? selectedDurationMs(),
      ) {
        stopQuizTimer();

        const mode = document.getElementById("timerMode").value;

        if (mode !== "countdown") {
          remainingTimeMs = null;
          renderTimer();
          return;
        }

        if (!quizStarted || quizExpired || quizFinished) {
          remainingTimeMs = durationMs;
          renderTimer();
          return;
        }

        remainingTimeMs = durationMs;
        timerDeadline = Date.now() + durationMs;

        renderTimer();

        timerInterval = setInterval(() => {
          remainingTimeMs = Math.max(0, timerDeadline - Date.now());

          if (remainingTimeMs <= 0) {
            remainingTimeMs = 0;
            stopQuizTimer();
            expireQuiz();
            return;
          }

          renderTimer();
        }, 200);
      }

      function handleTimerModeChange() {
        const mode = document.getElementById("timerMode").value;
        const durationLabel = document.getElementById("timerDurationLabel");

        // Show duration options only for countdown mode.
        durationLabel.style.display = mode === "countdown" ? "" : "none";

        // Stop any previously running timer when changing modes.
        stopQuizTimer();
        stopElapsedTimer();

        // The quiz has ended: don't allow another timer to start.
        if (quizFinished || quizExpired) {
          updateCountUpButton();
          renderTimer();
          return;
        }

        // Before START QUIZ, prepare the timer without starting it.
        if (!quizStarted) {
          remainingTimeMs = mode === "countdown" ? selectedDurationMs() : null;

          elapsedTimeMs = 0;

          renderElapsedTimer();
          renderTimer();
          updateCountUpButton();
          return;
        }

        // The quiz is already active.
        if (mode === "countdown") {
          remainingTimeMs = selectedDurationMs();
          startQuizTimer(remainingTimeMs);
        } else if (mode === "countup") {
          remainingTimeMs = null;
          startElapsedTimer();
        } else {
          remainingTimeMs = null;
        }

        renderTimer();
        updateCountUpButton();
      }

      function handleTimerDurationChange() {
        // Changing the duration only prepares the countdown.
        // It does not start the clock or interrupt a running quiz.
        if (!quizStarted || quizFinished || quizExpired) {
          remainingTimeMs =
            document.getElementById("timerMode").value === "countdown"
              ? selectedDurationMs()
              : null;

          renderTimer();
        }
      }

      function startQuiz() {
        if (quizStarted || quizExpired || quizFinished) return;

        quizStarted = true;

        document.getElementById("startPanel").style.display = "none";

        // Reveal the answer bank now that the quiz has started.
        renderQuiz();

        // // The count-up clock always starts, even if countdown is disabled.
        // startElapsedTimer();

        const mode = document.getElementById("timerMode").value;

        if (mode === "countdown") {
          startQuizTimer(remainingTimeMs ?? selectedDurationMs());
        } else if (mode === "countup") {
          startElapsedTimer();
          renderTimer();
        } else {
          renderTimer();
        }
      }

      function expireQuiz() {
        if (quizExpired || quizFinished) return;

        quizExpired = true;

        stopQuizTimer();

          "Time is up! Your quiz has ended.";

        renderTimer();
        finishQuiz();

      /*
       * ============================================================
       * SHUFFLE
       * ============================================================
       */

      function shuffle(array) {
        const copy = [...array];

        for (let i = copy.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));

          [copy[i], copy[j]] = [copy[j], copy[i]];
        }

        return copy;
      }

       * TOGGLE NORMAL / REVERSE QUIZ

      function toggleQuizMode() {
        if (quizExpired || quizFinished) return;

        const timerEnabled =
          document.getElementById("timerMode").value !== "off";
        const wasStarted = quizStarted;

        const preservedTime = timerEnabled
          ? wasStarted && timerDeadline !== null
            ? Math.max(0, timerDeadline - Date.now())
            : (remainingTimeMs ?? selectedDurationMs())
          : null;

        if (timerEnabled && wasStarted && preservedTime <= 0) {
          expireQuiz();
          return;
        }

        reverseMode = !reverseMode;

        if (!reviewMode) {
          newQuiz();

          if (timerEnabled) {
            remainingTimeMs = preservedTime;
            renderTimer();
          }

          return;
        }

        const remaining = currentQuiz.filter((item) =>
          missedMedications.has(medicationKey(item)),
        );

        if (remaining.length === 0) {
          reviewMode = false;
          newQuiz();

          if (timerEnabled) {
            remainingTimeMs = preservedTime;
            renderTimer();
          }

          return;
        }

        currentQuiz = shuffle(remaining);

        reviewFirstAttempt = new Set(
          [...reviewFirstAttempt].filter((key) =>
            currentQuiz.some((item) => medicationKey(item) === key),
          ),
        );

        // Changing direction resets progress and returns to the start screen.
        stopQuizTimer();
        stopElapsedTimer();

        quizStarted = false;
        elapsedTimeMs = 0;
        elapsedStartTime = null;
        selectedAnswer = null;
        score = 0;
        answered = 0;

        remainingTimeMs = timerEnabled ? preservedTime : null;

        document.getElementById("startPanel").style.display = "block";
        document.getElementById("total").textContent = currentQuiz.length;
        document.getElementById("completion").style.display = "none";
        document.getElementById("feedback").textContent = "";

        updateStats();
        updateModeText();
        renderQuiz();
        renderElapsedTimer();
        renderTimer();
      }

            ? "Drag each generic answer into the empty spot beside its matching brand name. On a phone, tap the answer and then its destination."
            : "Drag each brand answer into the empty spot beside its matching generic name. On a phone, tap the answer and then its destination.";

          return;
        }

        if (reverseMode) {
          description.textContent =
            "Match each brand-name medication with its correct generic name.";

          instruction.textContent =
            "Drag each generic name into the empty box beside its matching brand name. On a phone, tap a generic name and then tap its destination.";
        } else {
          description.textContent =
            "Match each generic medication with its correct brand name.";

          instruction.textContent =
            "Drag each brand name into the empty box beside its matching generic name. On a phone, tap a brand and then tap its destination.";
        }
      }

      function renderQuiz() {
        const grid = document.getElementById("quizGrid");

        grid.innerHTML = "";

        // Shuffle the answers independently from the questions.
        const randomizedAnswers = shuffle(currentQuiz);

        // Create the three columns.
        const questionColumn = document.createElement("div");
        questionColumn.className = "column";

        const dropColumn = document.createElement("div");
        dropColumn.className = "column";

        const answerColumn = document.createElement("div");

        answerColumn.className = quizStarted
          ? "column answer-column"
          : "column answer-column answers-hidden";

        // Column headings change with the quiz direction.
        const questionHeading = document.createElement("div");
        questionHeading.className = "column-heading";
        questionHeading.textContent = reverseMode
          ? "Brand Name"
          : "Generic Name";

        const dropHeading = document.createElement("div");
        dropHeading.className = "column-heading";
        dropHeading.textContent = "Your Answer";

        const answerHeading = document.createElement("div");
        answerHeading.className = "column-heading";
        answerHeading.textContent = reverseMode
          ? "Generic Answers"
          : "Brand Answers";

        questionColumn.appendChild(questionHeading);
        dropColumn.appendChild(dropHeading);
        answerColumn.appendChild(answerHeading);

        // COLUMN 1: Questions.
        currentQuiz.forEach((item) => {
          const question = document.createElement("div");
          question.className = "generic";

          question.textContent = reverseMode ? item.brand : item.generic;

          questionColumn.appendChild(question);
        });

        // COLUMN 2: Matching drop zones.
        currentQuiz.forEach((item) => {
          const drop = document.createElement("div");
          drop.className = "drop-zone";

          drop.textContent = "Drop answer here";

          // Store the correct answer for this question.
          drop.dataset.answer = reverseMode ? item.generic : item.brand;
          drop.dataset.medicationKey = medicationKey(item);

          setupDropZone(drop);

          dropColumn.appendChild(drop);
        });

        // COLUMN 3: Shared, independently randomized answer bank.
        randomizedAnswers.forEach((item) => {
          const answer = document.createElement("div");
          answer.className = "brand";

          answer.textContent = reverseMode ? item.generic : item.brand;

          answer.dataset.answer = reverseMode ? item.generic : item.brand;

          answer.draggable = true;

          setupAnswer(answer);

          answerColumn.appendChild(answer);
        });

        // Display the columns side by side.
        grid.appendChild(questionColumn);
        grid.appendChild(dropColumn);
        grid.appendChild(answerColumn);

        // TEMPORARY DIAGNOSTIC: verify all three quiz columns.
        console.log("Quiz column diagnostic:", {
          expectedQuestions: currentQuiz.length,
          questionColumnItems:
            questionColumn.querySelectorAll(".generic").length,
          dropZoneItems: dropColumn.querySelectorAll(".drop-zone").length,
          answerBankItems: answerColumn.querySelectorAll(".brand").length,
          duplicateAnswerLabels: [...answerColumn.querySelectorAll(".brand")]
            .map((el) => el.textContent.trim())
            .filter((label, index, labels) => labels.indexOf(label) !== index),
        });
      }

      /*
       * ============================================================
       * SETUP MOVABLE ANSWER
       * ============================================================
       */

      function setupAnswer(answer) {
        /*
         * Desktop drag.
         */

        answer.addEventListener("dragstart", (event) => {
          if (answer.classList.contains("used")) {
            event.preventDefault();
            return;
          }

          event.dataTransfer.setData("text/plain", answer.dataset.answer);

          event.dataTransfer.effectAllowed = "move";

          answer.classList.add("dragging");

          startAutoScroll();
        });

        answer.addEventListener("dragend", () => {
          answer.classList.remove("dragging");

          stopAutoScroll();
        });

        /*
         * Phone / touch selection.
         */

        answer.addEventListener("click", () => {
          if (answer.classList.contains("used")) {
            return;
          }

          document
            .querySelectorAll(".brand.selected")
            .forEach((el) => el.classList.remove("selected"));

          selectedAnswer = answer;

          answer.classList.add("selected");

          document.getElementById("feedback").textContent = reverseMode
            ? "Now tap the empty spot for this generic name."
            : "Now tap the empty spot for this brand.";

          document.getElementById("feedback").className = "feedback";
        });
      }

      /*
       * ============================================================
       * DROP ZONES
       * ============================================================
       */

      function setupDropZone(drop) {
        /*
         * Desktop dragover.
         */

        drop.addEventListener("dragover", (event) => {
          event.preventDefault();

          if (drop.classList.contains("correct")) {
            return;
          }

          drop.classList.add("drag-over");
        });

        drop.addEventListener("dragleave", () => {
          drop.classList.remove("drag-over");
        });

        /*
         * Desktop drop.
         */

        drop.addEventListener("drop", (event) => {
          event.preventDefault();

          drop.classList.remove("drag-over");

          stopAutoScroll();

          const answerName = event.dataTransfer.getData("text/plain");

          const answerElement = [...document.querySelectorAll(".brand")].find(
            (el) =>
              el.dataset.answer === answerName &&
              !el.classList.contains("used"),
          );

          if (answerElement) {
            checkAnswer(drop, answerElement);
          }
        });

        /*
         * Mobile tap destination.
         */

        drop.addEventListener("click", () => {
          if (!selectedAnswer) {
            return;
          }

          checkAnswer(drop, selectedAnswer);

          selectedAnswer = null;
        });
      }

      /*
       * ============================================================
       * CHECK ANSWER
       * ============================================================
       */

      function checkAnswer(drop, answerElement) {
        if (!quizStarted || quizExpired || quizFinished) return;
        if (drop.classList.contains("correct")) {
          return;
        }

        const selectedAnswerName = answerElement.dataset.answer;
        const correctAnswer = drop.dataset.answer;
        const questionKey = drop.dataset.medicationKey;

        const questionItem = currentQuiz.find(
          (item) => medicationKey(item) === questionKey,
        );

        if (selectedAnswerName === correctAnswer) {
          // CORRECT ANSWER

          drop.textContent = selectedAnswerName;
          drop.classList.remove("incorrect");
          drop.classList.add("correct");

          answerElement.classList.remove("selected");
          answerElement.classList.add("used");

          // Clear a missed medication only if this was its
          // first attempt during the current review round.
          if (reviewMode) {
            if (reviewFirstAttempt.has(questionKey)) {
              missedMedications.delete(questionKey);
            }

            reviewFirstAttempt.delete(questionKey);
          }

          score++;
          answered++;

          showFeedback("Correct!", true);

          updateStats();
          updateReviewButton();

          if (answered === currentQuiz.length) {
            finishQuiz();
          }
        } else {
          // INCORRECT ANSWER

          // Record the medication once, even after repeated mistakes.
          if (questionItem) {
            missedMedications.set(questionKey, questionItem);
          }

          // The first attempt has now been used up.
          if (reviewMode) {
            reviewFirstAttempt.delete(questionKey);
          }

          drop.classList.remove("incorrect");

          void drop.offsetWidth;

          drop.classList.add("incorrect");

          showFeedback(
            reverseMode
              ? "Not quite — try another generic name."
              : "Not quite — try another brand.",
            false,
          );

          updateStats();
          updateReviewButton();
        }
      }

      /*
       * ============================================================
       * FEEDBACK
       * ============================================================
       */

      function showFeedback(message, success) {
        const feedback = document.getElementById("feedback");

        feedback.textContent = message;

        feedback.className = success ? "feedback success" : "feedback error";
      }

      /*
       * ============================================================
       * STATS
       * ============================================================
       */

      function updateStats() {
        document.getElementById("score").textContent = score;

        document.getElementById("answered").textContent = answered;

        const accuracy =
          answered === 0 ? 0 : Math.round((score / answered) * 100);

        document.getElementById("accuracy").textContent = accuracy + "%";
      }

      /*
       * ============================================================
       * COMPLETE
       * ============================================================
       */

      function finishQuiz() {
        if (quizFinished) return;

        quizFinished = true;

        stopQuizTimer();
        stopElapsedTimer();

        // Remove the warning glow when the quiz is over.
        document.body.classList.remove("timer-warning");

        renderTimer();

        setTimeout(() => {
          const percentage = currentQuiz.length
            ? Math.round((score / currentQuiz.length) * 100)
            : 0;

          document.getElementById("finalScore").textContent =
            `You scored ${score} out of ${currentQuiz.length} (${percentage}%).`;

          document.getElementById("completion").style.display = "block";
        }, 300);
      }

      /*
       * ============================================================
       * RESET
       * ============================================================
       */

      function resetQuiz() {
        stopQuizTimer();
        stopElapsedTimer();

        quizStarted = false;
        elapsedTimeMs = 0;
        const stopButton = document.getElementById("stopElapsedButton");
        stopButton.style.display = "none";
        stopButton.textContent = "Stop Timer";
        stopButton.disabled = false;
        elapsedStartTime = null;

        document.getElementById("startPanel").style.display = "block";

        remainingTimeMs =
          document.getElementById("timerMode").value === "countdown"
            ? selectedDurationMs()
            : null;

        renderElapsedTimer();

        score = 0;
        answered = 0;
        selectedAnswer = null;

        quizExpired = false;
        quizFinished = false;

        document.getElementById("completion").style.display = "none";
        document.getElementById("feedback").textContent = "";

        if (reviewMode) {
          currentQuiz = currentQuiz.filter((item) =>
            missedMedications.has(medicationKey(item)),
          );

          if (currentQuiz.length === 0) {
            reviewMode = false;
            newQuiz();
            return;
          }
        }

        document.getElementById("total").textContent = currentQuiz.length;

        updateStats();
        updateReviewButton();
        renderQuiz();

        // startQuizTimer();
        renderTimer();
      }

      /*
       * ============================================================
       * AUTO-SCROLL WHILE DRAGGING
       * ============================================================
       */

      let autoScrollAnimation = null;

      let autoScrollSpeed = 0;

      function updateAutoScroll(clientY) {
        const edgeSize = 90;

        const maxSpeed = 14;

        const viewportHeight = window.innerHeight;

        autoScrollSpeed = 0;

        /*
         * Near top.
         */

        if (clientY < edgeSize) {
          const distance = edgeSize - clientY;

          autoScrollSpeed = -Math.min(maxSpeed, Math.max(2, distance / 4));
        } else if (clientY > viewportHeight - edgeSize) {
          /*
           * Near bottom.
           */
          const distance = clientY - (viewportHeight - edgeSize);

          autoScrollSpeed = Math.min(maxSpeed, Math.max(2, distance / 4));
        }

        if (autoScrollSpeed !== 0 && !autoScrollAnimation) {
          startAutoScroll();
        }
      }

      function startAutoScroll() {
        if (autoScrollAnimation) {
          return;
        }

        function scrollLoop() {
          if (autoScrollSpeed !== 0) {
            window.scrollBy(0, autoScrollSpeed);
          }

          autoScrollAnimation = requestAnimationFrame(scrollLoop);
        }

        autoScrollAnimation = requestAnimationFrame(scrollLoop);
      }

      function stopAutoScroll() {
        autoScrollSpeed = 0;

        if (autoScrollAnimation) {
          cancelAnimationFrame(autoScrollAnimation);

          autoScrollAnimation = null;
        }
      }

      /*
       * ============================================================
       * DESKTOP DRAG AUTO-SCROLL
       * ============================================================
       */

      document.addEventListener("dragover", (event) => {
        updateAutoScroll(event.clientY);
      });

      document.addEventListener("drop", () => {
        stopAutoScroll();
      });

      document.addEventListener("dragend", () => {
        stopAutoScroll();
      });

      });
